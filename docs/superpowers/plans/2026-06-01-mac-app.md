# KotaTask macOS — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Создать нативное macOS-приложение KotaTask в папке `mac/` как полный порт Android-версии, не затрагивая папку `android/`.

**Architecture:** MVVM. Данные — SwiftData (@Model классы). Навигация — NavigationSplitView (sidebar + список + детали). Алгоритм приоритета — чистая функция PriorityCalculator (легко тестируется). @Query в views для чтения данных, ModelContext через @Environment для записи.

**Tech Stack:** Swift 5.9+, SwiftUI 5, SwiftData, macOS 14+, UNUserNotificationCenter, XCTest

**Spec:** [`docs/superpowers/specs/2026-06-01-mac-app-design.md`](../specs/2026-06-01-mac-app-design.md)

- [ ] **Шаг 4: Реализовать PriorityCalculator.swift**

  ```swift
  import Foundation

  struct PriorityWeights {
      let urgency: Double
      let importance: Double
      let effort: Double
      let deadline: Double

      static let `default` = PriorityWeights(
          urgency: 0.35,
          importance: 0.30,
          effort: 0.20,
          deadline: 0.15
      )

      static func from(settings: AppSettings) -> PriorityWeights {
          PriorityWeights(
              urgency: settings.weightUrgency,
              importance: settings.weightImportance,
              effort: settings.weightEffort,
              deadline: settings.weightDeadline
          )
      }
  }

  struct PriorityCalculator {

      /// Рассчитывает приоритет задачи. Возвращает значение 0–100.
      static func calculate(
          urgency: Int,
          importance: Int,
          effort: Int,
          dueDate: Date?,
          weights: PriorityWeights
      ) -> Double {
          let urgencyScore    = Double(urgency) / 5.0
          let importanceScore = Double(importance) / 5.0
          let effortScore     = 1.0 - (Double(effort) / 5.0)
          let deadlineScore   = deadlineFactor(dueDate: dueDate)

          let raw = urgencyScore    * weights.urgency
                  + importanceScore * weights.importance
                  + effortScore     * weights.effort
                  + deadlineScore   * weights.deadline

          return (raw * 100).rounded()
      }

      /// Удобный вариант: принимает AppTask и AppSettings.
      static func calculate(task: AppTask, settings: AppSettings) -> Double {
          calculate(
              urgency: task.urgency,
              importance: task.importance,
              effort: task.effort,
              dueDate: task.dueDate,
              weights: PriorityWeights.from(settings: settings)
          )
      }

      /// Дедлайн-фактор: 0.0 (нет дедлайна) → 1.0 (просрочено).
      static func deadlineFactor(dueDate: Date?) -> Double {
          guard let due = dueDate else { return 0.0 }
          let days = Calendar.current.dateComponents([.day], from: .now, to: due).day ?? 0
          switch days {
          case ..<0:   return 1.0
          case 0...1:  return 0.9
          case 2...3:  return 0.7
          case 4...7:  return 0.5
          default:     return 0.2
          }
      }
  }
  ```

  import SwiftUI
  import SwiftData

  struct TaskRowView: View {
      @Bindable var task: AppTask
      @Environment(\.modelContext) private var context
      @Query private var settingsList: [AppSettings]

      private var settings: AppSettings? { settingsList.first }

      private var priorityColor: Color {
          switch task.priority {
          case 70...: return .blue
          case 40..<70: return .green
          default: return .orange
          }
      }

      var body: some View {
          HStack(spacing: 12) {
              // Чекбокс
              Button(action: toggleDone) {
                  Image(systemName: task.status == .done ? "checkmark.circle.fill" : "circle")
                      .foregroundColor(task.status == .done ? .green : .secondary)
                      .font(.system(size: 16))
              }
              .buttonStyle(.plain)

              // Текст
              VStack(alignment: .leading, spacing: 2) {
                  Text(task.title)
                      .strikethrough(task.status == .done)
                      .foregroundColor(task.status == .done ? .secondary : .primary)
                  if let due = task.dueDate {
                      Text(due, style: .date)
                          .font(.caption)
                          .foregroundColor(.secondary)
                  }
              }

              Spacer()

              // Число приоритета
              Text("\(Int(task.priority))")
                  .font(.system(size: 13, weight: .semibold, design: .rounded))
                  .foregroundColor(priorityColor)
                  .frame(minWidth: 30, alignment: .trailing)
          }
          .padding(.vertical, 4)
          .contentShape(Rectangle())
      }

      private func toggleDone() {
          if task.status == .done {
              task.status = .todo
              task.completedAt = nil
          } else {
              task.status = .done

  ```swift
  import Foundation
  import UserNotifications
  import SwiftData

  final class NotificationManager {
      static let shared = NotificationManager()
      private init() {}

      private let notificationId = "kotatask.daily"

      func requestPermission() {
          UNUserNotificationCenter.current().requestAuthorization(options: [.alert, .sound, .badge]) { granted, error in
              if let error { print("Notification permission error: \(error)") }
          }
      }

      /// Отменяет старое уведомление и планирует новое если включено.
      func reschedule(settings: AppSettings) {
          UNUserNotificationCenter.current().removePendingNotificationRequests(withIdentifiers: [notificationId])
          guard settings.notificationsEnabled else { return }
          schedule(hour: settings.notificationHour, minute: settings.notificationMinute)
      }

      /// Вызывать при старте приложения чтобы восстановить уведомление после перезагрузки.
      func restoreIfNeeded(settings: AppSettings) {
          guard settings.notificationsEnabled else { return }
          UNUserNotificationCenter.current().getPendingNotificationRequests { requests in
              if requests.first(where: { $0.identifier == self.notificationId }) == nil {
                  self.schedule(hour: settings.notificationHour, minute: settings.notificationMinute)
              }
          }
      }

      private func schedule(hour: Int, minute: Int) {
          let content = UNMutableNotificationContent()
          content.title = "KotaTask"
          content.body = "Время проверить свои задачи на сегодня"
          content.sound = .default

          var dateComponents = DateComponents()
          dateComponents.hour = hour
          dateComponents.minute = minute

          let trigger = UNCalendarNotificationTrigger(dateMatching: dateComponents, repeats: true)
          let request = UNNotificationRequest(identifier: notificationId, content: content, trigger: trigger)

          UNUserNotificationCenter.current().add(request) { error in
              if let error { print("Notification scheduling error: \(error)") }
          }
      }
  }
  ```
